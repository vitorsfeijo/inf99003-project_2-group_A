\section{Conclusão}
\begin{frame}{Limites da evidência}
\vspace{4mm}
\begin{tabular}{@{}p{.44\textwidth}p{.49\textwidth}@{}}
{\large Pacientes sintéticos} & Perfis e ausências simulados\\[6mm]
{\large Caminhada estimada} & Rede OSM/OSRM; sem medição em campo\\[6mm]
{\large Amostras controladas} & Uma equipe; poucas sementes\\[6mm]
{\large Comparação descritiva} & Sem superioridade universal
\end{tabular}
\smallnote{Eficiência simulada do algoritmo. Os resultados não medem benefício clínico na população.}
\end{frame}
% 19
\begin{frame}[c]{Conclusão}
\centering\bigtext{Mais prioridade atendida a tempo\\nas simulações com rotas a pé}
\vspace{7mm}{\large Fatorial: menor atraso, +3,03\% de caminhada}
\par\vspace{4mm}{\large 250 pacientes: 5,56 dias antes, 13,03\% menos caminhada}
\par\vspace{6mm}\labeltext{Ante o vizinho próximo · mais cálculo · resultados dependem do cenário}
\vspace{5mm}\par\labeltext{Artefato reutilizável + experimentos reproduzíveis}
\end{frame}
